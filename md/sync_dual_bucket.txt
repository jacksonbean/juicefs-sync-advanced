# 双 Bucket 并行迁移方案

## 方案概述

- **目标**：同时迁移 2 个 Bucket 到 COS
- **策略**：3 台机器并行，每台机器同时运行 2 个迁移任务
- **原则**：准确优先，存在即跳过，简单可重跑

---

## 架构设计

| 机器 | Bucket1 范围 | Bucket2 范围 |
|------|--------------|--------------|
| 机器1 | 0-3 开头 | 0-3 开头 |
| 机器2 | 4-6 开头 | 4-6 开头 |
| 机器3 | 7-9 开头 | 7-9 开头 |

每台机器同时运行 2 个迁移任务（前后台运行）。

---

## 准备工作

在三台机器上分别执行：

```bash
# 创建必要目录
mkdir -p /opt/migration /var/log/juicefs/bucket1 /var/log/juicefs/bucket2

# 添加执行权限
chmod 755 /opt/migration /var/log/juicefs
```

## 机器迁移脚本

创建 `/opt/migration/migrate.sh`：

```bash
#!/bin/bash
# migrate.sh - 双 Bucket 并行迁移脚本

# 迁移任务函数
run_migration() {
    local RANGE=$1
    local BUCKET_NAME=$2
    local SRC_URL=$3
    local DST_URL=$4
    local LOG_DIR=$5

    local FAILED_LIST="$LOG_DIR/failed_${RANGE}.list"

    mkdir -p $LOG_DIR
    > $FAILED_LIST

    echo "===== 开始迁移 $BUCKET_NAME $RANGE 开头文件 $(date) =====" | tee -a $LOG_DIR/migration.log

    # 后台实时监控并提取错误
    (
        tail -F $LOG_DIR/juicefs_${RANGE}.log 2>/dev/null | while read line; do
            if echo "$line" | grep -qiE "error|fail|timeout"; then
                echo "$line" | grep -oE 's3://[a-zA-Z0-9._/-]+' >> "$FAILED_LIST"
                echo "[$(date '+%Y-%m-%d %H:%M:%S')] $line" >> "$FAILED_LIST.full"
            fi
        done
    ) &
    local TAIL_PID=$!

    local START_TIME=$(date +%s)

    # 根据范围设置 start/end
    case $RANGE in
        "0-3")  START_KEY="0"; END_KEY="4" ;;
        "4-6")  START_KEY="4"; END_KEY="7" ;;
        "7-9")  START_KEY="7"; END_KEY=":" ;;
    esac

    juicefs sync \
        --ignore-existing \
        --threads 200 \
        --list-threads 50 \
        --start "$START_KEY" --end "$END_KEY" \
        --max-failure 10 \
        --check-change \
        "$SRC_URL" \
        "$DST_URL" \
        2>&1 | tee "$LOG_DIR/juicefs_${RANGE}.log"

    local EXIT_CODE=${PIPESTATUS[0]}
    kill $TAIL_PID 2>/dev/null

    local END_TIME=$(date +%s)
    local DURATION=$((END_TIME - START_TIME))

    local COPIED=$(grep -c "Copied" "$LOG_DIR/juicefs_${RANGE}.log" 2>/dev/null || echo 0)
    local SKIPPED=$(grep -c -i "exist" "$LOG_DIR/juicefs_${RANGE}.log" 2>/dev/null || echo 0)
    local ERRORS=$(grep -icE "error|fail" "$LOG_DIR/juicefs_${RANGE}.log" 2>/dev/null || echo 0)

    # 去重失败列表
    if [ -f "$FAILED_LIST" ]; then
        sort -u "$FAILED_LIST" -o "$FAILED_LIST"
        local failed_count=$(wc -l < "$FAILED_LIST")
        [ "$failed_count" -gt 0 ] && echo "⚠️ 发现 $failed_count 个失败文件" | tee -a $LOG_DIR/migration.log
    fi

    echo "===== $BUCKET_NAME $RANGE 开头文件迁移完成 $(date) =====" | tee -a $LOG_DIR/migration.log
    echo "耗时: ${DURATION}秒 | 已复制: $COPIED | 已跳过: $SKIPPED | 错误: $ERRORS" | tee -a $LOG_DIR/migration.log
    echo "退出码: $EXIT_CODE" | tee -a $LOG_DIR/migration.log

    return $EXIT_CODE
}

# ============ 配置区域 - 根据实际情况修改 ============

# Bucket1 配置
BUCKET1_NAME="bucket1"
BUCKET1_SRC="s3://ecs-bucket1/"      # 源 Bucket1 URL
BUCKET1_DST="cos://cos-bucket1/"     # 目标 Bucket1 URL

# Bucket2 配置
BUCKET2_NAME="bucket2"
BUCKET2_SRC="s3://ecs-bucket2/"      # 源 Bucket2 URL
BUCKET2_DST="cos://cos-bucket2/"     # 目标 Bucket2 URL

# 日志目录
LOG_DIR="/var/log/juicefs"

# 本机的迁移范围（根据机器编号选择）
# 机器1: 0-3, 机器2: 4-6, 机器3: 7-9
MACHINE_RANGE=${1:-"0-3"}  # 可通过参数指定，或直接修改这里

# ====================================================

LOG_DIR_B1="$LOG_DIR/bucket1"
LOG_DIR_B2="$LOG_DIR/bucket2"

case $MACHINE_RANGE in
    "0-3")
        echo ">>> 机器执行范围: 0-3 开头"
        # 后台运行 bucket1
        run_migration "0-3" "$BUCKET1_NAME" "$BUCKET1_SRC" "$BUCKET1_DST" "$LOG_DIR_B1" &
        PID_B1=$!

        # 前台运行 bucket2
        run_migration "0-3" "$BUCKET2_NAME" "$BUCKET2_SRC" "$BUCKET2_DST" "$LOG_DIR_B2"
        EXIT_B2=$?

        # 等待 bucket1
        wait $PID_B1
        EXIT_B1=$?
        ;;
    "4-6")
        echo ">>> 机器执行范围: 4-6 开头"
        run_migration "4-6" "$BUCKET1_NAME" "$BUCKET1_SRC" "$BUCKET1_DST" "$LOG_DIR_B1" &
        PID_B1=$!

        run_migration "4-6" "$BUCKET2_NAME" "$BUCKET2_SRC" "$BUCKET2_DST" "$LOG_DIR_B2"
        EXIT_B2=$?

        wait $PID_B1
        EXIT_B1=$?
        ;;
    "7-9")
        echo ">>> 机器执行范围: 7-9 开头"
        run_migration "7-9" "$BUCKET1_NAME" "$BUCKET1_SRC" "$BUCKET1_DST" "$LOG_DIR_B1" &
        PID_B1=$!

        run_migration "7-9" "$BUCKET2_NAME" "$BUCKET2_SRC" "$BUCKET2_DST" "$LOG_DIR_B2"
        EXIT_B2=$?

        wait $PID_B1
        EXIT_B1=$?
        ;;
    *)
        echo "未知范围: $MACHINE_RANGE"
        exit 1
        ;;
esac

echo ""
echo "===== 双 Bucket 迁移完成 ====="
echo "Bucket1 ($MACHINE_RANGE) 退出码: $EXIT_B1"
echo "Bucket2 ($MACHINE_RANGE) 退出码: $EXIT_B2"

# 任一任务失败则整体失败
[ $EXIT_B1 -ne 0 ] || [ $EXIT_B2 -ne 0 ] && exit 1
exit 0
```

## 简化版脚本（每台机器固定范围）

如果你不想用参数，也可以创建三个独立的脚本。

### 机器1: `/opt/migration/machine1.sh`

```bash
#!/bin/bash
# machine1.sh - 机器1：迁移 0-3 开头文件

LOG_DIR="/var/log/juicefs"
B1_LOG="$LOG_DIR/bucket1"
B2_LOG="$LOG_DIR/bucket2"

echo "===== 机器1开始迁移 0-3 开头 $(date) ====="

# 后台运行 bucket1
(
    juicefs sync --ignore-existing --threads 200 --list-threads 50 \
        --start "0" --end "4" --max-failure 10 --check-change \
        s3://ecs-bucket1/ cos://cos-bucket1/ \
        2>&1 | tee "$B1_LOG/juicefs_0-3.log"
) &
PID_B1=$!

# 前台运行 bucket2
juicefs sync --ignore-existing --threads 200 --list-threads 50 \
    --start "0" --end "4" --max-failure 10 --check-change \
    s3://ecs-bucket2/ cos://cos-bucket2/ \
    2>&1 | tee "$B2_LOG/juicefs_0-3.log"

EXIT_B2=$?
wait $PID_B1
EXIT_B1=$?

echo "===== 机器1完成 $(date) ====="
echo "Bucket1: $EXIT_B1 | Bucket2: $EXIT_B2"
[ $EXIT_B1 -ne 0 ] || [ $EXIT_B2 -ne 0 ] && exit 1
```

## 监控脚本

创建 `/opt/migration/monitor.sh`：

```bash
#!/bin/bash
# monitor.sh - 实时监控两个 Bucket 的迁移进度

LOG_DIR="/var/log/juicefs"

while true; do
    clear
    echo "==== 双 Bucket 迁移监控 - $(date) ===="
    echo ""

    for bucket in bucket1 bucket2; do
        echo "=== $bucket ==="
        BUCKET_LOG="$LOG_DIR/$bucket"

        for log in $BUCKET_LOG/juicefs_*.log; do
            [ -f "$log" ] || continue

            range=$(basename $log | grep -oE '[0-9]+-[0-9]+' || echo "未知")
            echo "--- 范围: $range ---"

            copied=$(grep -c "Copied" $log 2>/dev/null || echo 0)
            skipped=$(grep -c -i "exist" $log 2>/dev/null || echo 0)
            errors=$(grep -icE "error|fail" $log 2>/dev/null || echo 0)

            echo "已复制: $copied | 已跳过: $skipped | 错误: $errors"

            # 显示最近错误
            if [ "$errors" -gt 0 ]; then
                echo "最近错误:"
                grep -iE "error|fail" $log 2>/dev/null | tail -2 | sed 's/^/  /'
            fi
            echo ""
        done
    done

    # 系统资源
    echo "--- 系统状态 ---"
    echo "内存: $(free -h | awk '/Mem/{print $3"/"$2}')"
    echo "负载: $(uptime | awk -F'load average:' '{print $2}')"
    echo "磁盘: $(df -h /var | tail -1 | awk '{print $3"/"$2 " (" $5 ")"}')"

    # 失败文件
    for failed in $LOG_DIR/bucket*/failed_*.list; do
        [ -f "$failed" ] || continue
        count=$(wc -l < "$failed" 2>/dev/null || echo 0)
        [ "$count" -gt 0 ] && echo "待重试: $failed 有 $count 个"
    done

    echo ""
    echo "按 Ctrl+C 退出"
    sleep 30
done
```

## 执行步骤

### 1. 准备阶段

```bash
# 在三台机器上创建目录
mkdir -p /opt/migration /var/log/juicefs/bucket1 /var/log/juicefs/bucket2

# 复制脚本并添加执行权限
chmod +x /opt/migration/*.sh
```

### 2. 修改配置

编辑脚本中的 Bucket URL 配置：
- `BUCKET1_SRC`: 第一个源 Bucket 地址
- `BUCKET1_DST`: 第一个目标 Bucket 地址
- `BUCKET2_SRC`: 第二个源 Bucket 地址
- `BUCKET2_DST`: 第二个目标 Bucket 地址

### 3. 启动迁移

```bash
# 机器1
nohup /opt/migration/machine1.sh > /var/log/juicefs/machine1.out 2>&1 &

# 机器2
nohup /opt/migration/machine2.sh > /var/log/juicefs/machine2.out 2>&1 &

# 机器3
nohup /opt/migration/machine3.sh > /var/log/juicefs/machine3.out 2>&1 &
```

### 4. 查看进度

```bash
/opt/migration/monitor.sh
```
